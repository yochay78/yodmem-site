כל השירים

לילדים שלי היפים והטובים

לַיְּלָדִים שֶׁלִּי הַיָּפִים וְהַטּוֹבִים
בֶּאֱמֶת

יָעֵל אֲנָשִׁים כָּמוֹנוּ צְרִיכִים לַעֲמֹד עַל הָרַגְלַיִם.

אֲנָשִׁים כָּמוֹנוּ צְרִיכִים לְדַבֵּר.
זוֹ הַמְּשִׂימָה שֶׁלָּנוּ זֶה הָאֶתְגָּר.
אֲנָשִׁים חַלָּשִׁים הֵם הֲכִי חֲזָקִים.
רֹב הָאֲנָשִׁים חֲזָקִים אֲבָל זֶה לֹא
בֶּאֱמֶת בֶּאֱמֶת הֵם הֲכִי חַלָּשִׁים
לָכֵן הֵם עוֹטִים בֶּגֶד שִׁקְרִי שֶׁל חֹזְקָה מְדֻמָּה
זֶה לֹא בֶּאֱמֶת יָעֵל זֶה בִּגְלַל שֶׁהֵם
מְפַחֲדִים מֵאִתָּנוּ כִּי יֵשׁ לָנוּ בֶּאֱמֶת אִישִׁיּוּת
וְגַם יֵשׁ לָנוּ
עֵינַיִם.

אֲנַחְנוּ נִלְמַד לְדַבֵּר יָעֵל.
אֲנַחְנוּ נִצְעַק מִסּוֹף הָעוֹלָם
עַד סוֹף הָעוֹלָם כִּי רַק כָּכָה
הֵם מְבִינִים וְאָז נְדַבֵּר מַמָּשׁ
בְּשֶׁקֶט שֶׁיִּתְאַמְּצוּ לִשְׁמֹעַ
שֶׁיַּתְחִילוּ בֶּאֱמֶת לְהַקְשִׁיב
וְהֵם יַהַפְכוּ אוֹתָנוּ לֵאלֹהִים
וַאֲנַחְנוּ לֹא אֱלֹהִים
אֲנַחְנוּ רַק אֲנָשִׁים תְּמִימִים וְטוֹבִים
כְּמוֹ קַפְּטֵיין פּוֹל מַאֲמִינִים שֶׁכֻּלָּם
כָּמוֹנוּ אֲבָל כֻּלָּם מְשַׁקְּרִים יָעֵל
כֻּלָּם מְרַמִּים אָז אֵין לָנוּ
בְּרֵרָה לְצַעֲרֵנוּ אֲנַחְנוּ
צְרִיכִים גַּם לִלְמֹד
אֶת הַדְּרָכִים הָרָעוֹת
שֶׁלָּהֶם אֲבָל אֲנַחְנוּ נִהְיֶה
טוֹבִים אֲנַחְנוּ נַמְשִׁיךְ לִהְיוֹת
טוֹבִים.

לֹא עָנִית לִי הַאִם לָבוֹא אִתְּךָ מָחָר
לְטִפּוּל שִׁנַּיִם.

רֹב הָאֲנָשִׁים שׂוֹרְדִים
מְעַט הָאֲנָשִׁים חַיִּים
אֲנִי רוֹצֶה לִחְיוֹת
אַתְּ רוֹצָה
לִשְׂרֹד.

לִפְעָמִים אֵין בְּרֵרָה אֶלָּא לִשְׂרֹד.
לִשְׂרֹד עוֹד יוֹם לִשְׂרֹד עוֹד
לַיְלָה לִשְׂרֹד עוֹד שָׁעָה
לִשְׂרֹד עוֹד דַּקָּה
עוֹד שְׁנִיָּה.
רֹב הָאֲנָשִׁים לֹא מְסֻגָּלִים
לַחְשֹׁב שֶׁהֵם שׂוֹרְדִים
אָז הֵם חוֹשְׁבִים שֶׁהֵם
חֲזָקִים וְיֵשׁ לָהֶם פֵּרוּשִׁים
לוֹגִיִּים מַדְהִימִים לְשֵׁם כָּךְ.
נוּ, רֹב הָאֲנָשִׁים גַּם טִפְּשִׁים.
הֵם לֹא מְבִינִים מַה זֶּה לִהְיוֹת
חָכָם, בֶּאֱמֶת.

הֵם בּוֹנִים סִפּוּר
הֵם קוֹרְאִים לָזֶה אֱמֶת
וְהֵם נֶאֱחָזִים בָּהּ בְּצִפָּרְנַיִם
עַד דָּם.

הֵם מְדַבְּרִים חָזָק
כִּי אֵין לָהֶם מָה לְהַגִּיד

לִהְיוֹת חָכָם זֶה לִהְיוֹת אַמִּיץ
לִהְיוֹת אַמִּיץ זֶה לִהְיוֹת אֲמִתִּי
לִהְיוֹת אֲמִתִּי זֶה לִחְיוֹת
לִחְיוֹת זֶה לִחְיוֹת
בֶּאֱמֶת.

אֲנָשִׁים אוֹהֲבִים לִהְיוֹת בָּאֶמְצַע
בַּמָּקוֹם הַבָּטוּחַ בְּאֶמְצַע הָעֵדֶר
אַף אֶחָד לֹא רוֹצֶה לִהְיוֹת עֵז תּוֹעָה
מְחַפֶּשֶׂת אֶת הַדֶּרֶךְ שֶׁלָּהּ.
עֵדֶר כְּבָשִׂים עֵינַיִם לְמַטָּה
לָשׁוֹן בַּיַּשְׁבָן שֶׁל כֶּבֶשׂ אַחֵר.

הָאֲנָשִׁים שֶׁמִּקְּדִימָה
הָאֲנָשִׁים מֵאָחוֹר
הָאֲנָשִׁים בְּצַד שְׂמֹאל
הָאֲנָשִׁים בְּיָמִין
כֻּלָּם אוֹתָם אֲנָשִׁים
הֵם לֹא מְלַקְּקִים לְאַף אֶחָד
אֶת הַתַּחַת.

תְּלוּת זֶה הַדָּבָר הֲכִי גָּרוּעַ שֶׁיֵּשׁ
אַף אֶחָד לֹא חַיָּב לְאַף אֶחָד
כְּלוּם
אַף אֶחָד לֹא שַׁיָּךְ לְאַף אֶחָד
שְׁלִיטָה זֶה מַגְעִיל.

מִכֵּיוָן שֶׁרָאִיתִי אֶת הָעוֹלָם
כֹּה מְדֻרְדָּר הִצְטָרַכְתִּי
לוֹמַר אֶת הַמִּלִּים
הָאֵלֶּה וּמִכֵּיוָן שֶׁהָעוֹלָם
חֵרֵשׁ בִּגְלַל מוּזִיקַת
הָרֶקַע הַבִּלְתִּי פּוֹסֶקֶת
שֶׁאֵין לָהּ נְשָׁמָה
וְאֵין לָהּ אִישִׁיּוּת
הִצְטָרַכְתִּי לִצְעֹק
אוֹתָן לְתוֹךְ הָאָזְנַיִם
שֶׁלָּהֶם וַאֲנִי בִּכְלָל
שׂוֹנֵא לְדַבֵּר.
