וְהִנֵּה בַּיִת וְזֶה
דֶּלֶת לִפְתּוֹחַ
וְדֶלֶת לִסְגּוֹר
כְּמוֹ לֵב

וּבִפְנֵי אֲנָשִׁים
וַאֲנָשִׁים הֵם אוֹרְחִים
וְאַף אַתָּה
אוֹרֵחַ

וְאַף אַתָּה אִישׁ
וְאַתָּה יוֹשֵׁב בַּבַּיִת
וְאַתָּה לְבַד
וְאַתָּה מְעַשֵּׁן
וַאֲנָשִׁים צְרִיכִים
בַּיִת
וַאֲנָשִׁים צְרִיכִים
לֵב

וּפַעַם בָּאָה אֵלַיי
בְּלֵיל גֶּשֶׁם
רְטוּבָּה וְנִרְגֶּשֶׁת
וּפָתַחְתִּי הַדֶּלֶת

וְהַלֵּב מְפַחֵד מְאוֹד
וְאֵיךְ שֶׁפּוֹתֵחַ יָשָׁר
סוֹגֵר
כְּמוֹ מַלְכּוֹדֶת עַכְבָּרִים

וְיֵשׁ לֵב שֶׁפּוֹתֵחַ
וּפוֹתֵחַ וּמְגַלֶּה עֶרְוָותוֹ
וְכוּלָּם רוֹאִים לוֹ וְצוֹחֲקִים
הִנֵּה לֵב

וְטוֹב לָאָדָם לַעֲשׂוֹת
הַלֵּב בְּצֶלֶם בֵּיתוֹ
וְיִהְיֶה חַם וּמוּאָר וְדֶלֶת
לְהִיכָּנֵס וְדֶלֶת לָצֵאת וְחַלּוֹן

וּכְשֶׁחוֹזֵר בָּעֶרֶב
לְבֵיתוֹ וְהָעוֹלָם חָשׁוּךְ
טוֹב שֶׁיִּהְיֶה
שָׁם לֵב

וְטוֹב לָאָדָם לִבְנוֹת בַּיִת
בְּצֶלֶם הַלֵּב
עִם חֶדֶר יָמִין וְחֶדֶר שְׂמֹאל
וַעֲלִיָּה יָמִין וַעֲלִיָּה שְׂמֹאל

וְלַלֵּב יֵשׁ כִּיס
וְלַבַּיִת יֵשׁ גַּג
וְלָהּ לֹא הָיָה
אָז נָתְנוּ לָהּ כַּדּוּרִים

וְכִיס מִלְּשׁוֹן כִּיסּוּי
וְלֵב חַיָּיב קְצָת כִּיסּוּי
שֶׁלֹּא יִרְאוּ לוֹ
וּבַיִת גַּם כֵּן

וְאַרְבָּעָה חֲדָרִים
תַּחְתּוֹן וְעֶלְיוֹן כְּנֶגֶד אֶחָד
וְתַחְתּוֹן וְעֶלְיוֹן כְּנֶגֶד אַחַת
וּכְדַאי גַּם מִסְדְּרוֹן

וִינְשָּׁקוּ עֶלְיוֹן וְעֶלְיוֹן
וִינְשָּׁקוּ תַּחְתּוֹן וְתַחְתּוֹן
וִינְשָּׁקוּ עֶלְיוֹן וְתַחְתּוֹן
וִינְשָּׁקוּ תַּחְתּוֹן וְעֶלְיוֹן

וְאַרְבַּע נְשִׁיקוֹת
שֶׁל הַלֵּב
וְהַבַּיִת וְאָדָם
שֶׁהוּא שְׁנַיִים

וְאָדָם שֶׁרָחוֹק מְאוֹד מִלִּבּוֹ
טוֹב שֶׁיִּמְצָא לוֹ בַּיִת אַחֵר
וְאָדָם שֶׁרָחוֹק מִבֵּיתוֹ
צָרִיךְ לִמְצוֹא לֵב אַחֵר

וּבַהַתְחָלָה הַלֵּב נִשְׁבָּר
אֲבָל לֹא שָׂמִים לֵב
כִּי הוּא קָפוּא
וּכְשֶׁהַבַּיִת נִשְׁבָּר שָׂמִים לֵב

וְלָהּ נִשְׁבַּר הַבַּיִת
לִפְנֵי הַלֵּב
וְחִיפְּשָׂה בַּיִת בְּצוּרַת לֵב
וְקָנִיתִי מִמֶּנָּה גַּפְרוּר

וְקָשֶׁה לִמְצוֹא לֵב
וּבַיִת אֲמִיתִּיִּים
וְנֶחְלְקוּ חֲכָמִים
אִם מוּתָּר לְהִתְפַּשֵּׁר

וְאִם מָצָאתָ בַּיִת אוֹ לֵב
לֹא מָצָאתָ
וּמִצְוָוה לִזְכּוֹר תָּמִיד
שֶׁכְּלוּם לֹא שַׁיָּיךְ לְךָ

וַאֲפִילּוּ הַלֵּב
לֹא בֶּאֱמֶת שֶׁלְּךָ
קַל וָחוֹמֶר הַבַּיִת
וְחוֹמֶר הַגּוּף

וּכְשֶׁיּוֹצֵא אָדָם חוּצָה
יִלְבַּשׁ אֶת בֵּיתוֹ
מֵעַל מְעִילוֹ
וְייִקַּח אֶת הַלֵּב גַּם כֵּן

וְתָמִיד כְּשֶׁנִּשְׁבָּר הַבַּיִת
נִשְׁבָּר הַלֵּב
וְגַם לֶאֱלוֹהִים הָיָה בַּיִת
וְנִשְׁבַּר
